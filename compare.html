Early Theme Detection Script

Must appear BEFORE CSS loads to prevent flash of wrong theme:

<script>
    (function() {
        if (localStorage.getItem('reader_theme') === 'dark') {
            document.documentElement.setAttribute('data-theme', 'dark');
        }
    })();
</script>

Auth Check Script

Required for protected pages (appears in body, top):

<script>
    (function() {
        var session = localStorage.getItem('otr_beta_session');
        if (!session) {
            window.location.replace('./');
            return;
        }
        try {
            var data = JSON.parse(session);
            if (!data.authenticated || Date.now() > data.expires) {
                localStorage.removeItem('otr_beta_session');
                window.location.replace('./');
            }
        } catch (e) {
            window.location.replace('./');
        }
    })();
</script>
